### S1 补丁清单（核心产物）

```powershell
$k='G:\GTA\kelvaroq-main'; $out='G:\GTA\kelvaroq-eval'
New-Item -ItemType Directory -Force -Path $out | Out-Null
$rows = @()
foreach ($p in (Get-ChildItem $k -Recurse -Filter *.patch | Sort-Object Name)) {
  $head = Get-Content $p.FullName -TotalCount 40
  $targets = @($head | Select-String -Pattern '^(---|\+\+\+) [ab]/(.+)$' | ForEach-Object { ($_.Line -split '\s+')[1] -replace '^[ab]/','' } | Sort-Object -Unique)
  $lines = (Get-Content $p.FullName | Measure-Object -Line).Lines
  $subject = @($head | Select-String -Pattern '^Subject:|^# |^## ' | Select-Object -First 1 | ForEach-Object { $_.Line.Trim() })
  $rows += [pscustomobject]@{ patch=$p.Name; lines=$lines; targets=($targets -join ' '); subject=($subject -join ' ') }
}
$rows | Format-Table patch,lines,targets -AutoSize | Out-String -Width 200
"PATCH_COUNT=$($rows.Count)"
```
