## 命令

```powershell
# 0) 版本确认 + 额度守卫（门禁是额度，不是文件是否存在）
"SHA256=" + (Get-FileHash '<ROOT>\cool_fox\cool_fox-main\target\release\victory.exe' -Algorithm SHA256).Hash
$c = (Get-Counter '\Memory\Commit Limit','\Memory\Committed Bytes' -SampleInterval 1 -MaxSamples 1).CounterSamples
$limit = ($c | Where-Object Path -match 'commit limit').CookedValue / 1GB
$used  = ($c | Where-Object Path -match 'committed bytes').CookedValue / 1GB
$free  = [math]::Round($limit - $used, 2)
"commit_limit_gb=$([math]::Round($limit,2)) commit_used_gb=$([math]::Round($used,2)) commit_free_gb=$free"

if ($free -lt 12) {
  "SKIP_S5: commit_free_gb=$free < 12 —— 附本行原文回填，等页面文件扩完重投本任务"
} else {
  Get-Process | Where-Object { $_.ProcessName -match '^victory' } | Stop-Process -Force -ErrorAction SilentlyContinue
  Start-Sleep 3
  cd <ROOT>\cool_fox-优化方案
  powershell.exe -NoProfile -ExecutionPolicy Bypass -File .\13-bench-ab.ps1 -Label R3-A2-baseline-v2 -Exe "<ROOT>\cool_fox\cool_fox-main\target\release\victory-baseline.exe" -Seconds 200 -Profile fork
  Get-Process | Where-Object { $_.ProcessName -match '^victory' } | Stop-Process -Force -ErrorAction SilentlyContinue
  Start-Sleep 5
  powershell.exe -NoProfile -ExecutionPolicy Bypass -File .\13-bench-ab.ps1 -Label R3-B-reverted-v2 -Exe "<ROOT>\cool_fox\cool_fox-main\target\release\victory.exe" -Seconds 200 -Profile fork
  Get-Process | Where-Object { $_.ProcessName -match '^victory' } | Stop-Process -Force -ErrorAction SilentlyContinue
  "--- bench-v2.csv 全文 ---"
  $n = 0; Get-Content .\data\bench-v2.csv | ForEach-Object { $n++; "{0,2}: {1}" -f $n, $_ }
}
```
