  "=== $p ==="

## S2 复制存档/设置（只增不删，不覆盖 fork 已有文件）

```powershell
$ours='G:\GTA\victory-data\coolfox-data'; $fork='G:\GTA\victory-data\fork-eval-data'
New-Item -ItemType Directory -Force -Path $fork | Out-Null
Get-ChildItem $ours -Directory -ErrorAction SilentlyContinue | ForEach-Object {
  $dst = Join-Path $fork $_.Name
  if (-not (Test-Path $dst)) { Copy-Item $_.FullName $dst -Recurse -Force; "copied_dir=" + $_.Name }
  else { "exists_dir=" + $_.Name }
}
Get-ChildItem $ours -File -ErrorAction SilentlyContinue | ForEach-Object {
  $dst = Join-Path $fork $_.Name
  if (-not (Test-Path $dst)) { Copy-Item $_.FullName $dst -Force; "copied_file=" + $_.Name } else { "exists_file=" + $_.Name }
}
"fork_files_after=" + @(Get-ChildItem $fork -Recurse -File -ErrorAction SilentlyContinue).Count
```
