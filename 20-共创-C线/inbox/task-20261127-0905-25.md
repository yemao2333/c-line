### S1 生成候选清单（只读）

```powershell
$O='G:\GTA\cool_fox\cool_fox-main'
$all = @(Get-ChildItem $O -Recurse -Include *.rs -ErrorAction SilentlyContinue |
  Select-String -Pattern 'VICTORY_[A-Z0-9_]+' -AllMatches |
  ForEach-Object { $_.Matches } | ForEach-Object { $_.Value } | Sort-Object -Unique)
$hot = @($all | Where-Object { $_ -match 'COPY|BARRIER|DESCRIPTOR|HEAP|READBACK|STAGE|RANGE|BATCH|SUBMIT|REFRESH|CARD' })
$used = @()
# 已在启动脚本/实验脚本里出现过的算「已测/已用」
foreach ($f in @('G:\GTA\cool_fox\cool_fox-main\Start-GTA6.cmd','G:\GTA\cool_fox-优化方案\13-bench-ab.ps1','G:\GTA\cool_fox-优化方案\60-项目D\run-shot.ps1')) {
  if (Test-Path $f) { $used += @(Select-String -Path $f -Pattern 'VICTORY_[A-Z0-9_]+' -AllMatches | ForEach-Object { $_.Matches } | ForEach-Object { $_.Value }) }
}
$cand = @($hot | Where-Object { $used -notcontains $_ } | Sort-Object)
"all=$($all.Count) hot=$($hot.Count) used=$(@($used|Sort-Object -Unique).Count) cand=$($cand.Count)"
$cand -join "`n"
```
